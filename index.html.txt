    function sendToLine() {
        const statusDiv = document.getElementById('status');
        
        if (!navigator.geolocation) {
            statusDiv.innerHTML = "抱歉，您的手機瀏覽器不支援定位功能。";
            return;
        }

        statusDiv.innerHTML = "正在取得您的目前位置，請稍候...";

        navigator.geolocation.getCurrentPosition(
            (position) => {
                const lat = position.coords.latitude;
                const lon = position.coords.longitude;
                
                // 產生 Google Maps 導航連結
                const mapsUrl = `https://www.google.com/maps?q=${lat},${lon}`;
                const message = `🚨 您的寵物在這裡被發現了！\n📍 好心人位置：\n${mapsUrl}`;

                statusDiv.innerHTML = "即將開啟 LINE 傳送位置...";

                // 使用 LINE 官方文字分享協定
                const lineShareUrl = `https://line.me/R/msg/text/?${encodeURIComponent(message)}`;
                
                setTimeout(() => {
                    window.location.href = lineShareUrl;
                }, 1000);
            },
            (error) => {
                switch(error.code) {
                    case error.PERMISSION_DENIED:
                        statusDiv.innerHTML = "❌ 您取消了定位權限。請允許定位才能通知主人喔！";
                        break;
                    default:
                        statusDiv.innerHTML = "❌ 無法取得位置，請確認手機 GPS 是否開啟，或直接撥打電話。";
                        break;
                }
            },
            { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
        );
    }
</script>
